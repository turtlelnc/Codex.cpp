cmake_minimum_required(VERSION 3.16)
project(CodexCpp LANGUAGES CXX)
option(CODEX_WITH_GITX "Experimental embedded Gitx backend for local web diffs" OFF)
set(CODEX_GITX_SOURCE_DIR "" CACHE PATH "Path to the Gitx source checkout")
find_package(Threads REQUIRED)
add_executable(codex-cpp codex.cpp)
target_compile_features(codex-cpp PRIVATE cxx_std_17)
set_target_properties(codex-cpp PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(codex-cpp PRIVATE Threads::Threads)
if(CODEX_WITH_GITX)
  enable_language(C)
  if(NOT EXISTS "${CODEX_GITX_SOURCE_DIR}/src/git_repository.cpp")
    message(FATAL_ERROR "Set CODEX_GITX_SOURCE_DIR to a Gitx checkout")
  endif()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_CLI OFF CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(USE_SSH OFF CACHE STRING "" FORCE)
  set(USE_HTTPS OFF CACHE STRING "" FORCE)
  add_subdirectory("${CODEX_GITX_SOURCE_DIR}/third_party/libgit2-1.8.4" gitx-libgit2 EXCLUDE_FROM_ALL)
  add_library(codex-gitx-core STATIC
    "${CODEX_GITX_SOURCE_DIR}/src/git_repository.cpp"
    "${CODEX_GITX_SOURCE_DIR}/src/config.cpp")
  target_compile_features(codex-gitx-core PRIVATE cxx_std_20)
  target_include_directories(codex-gitx-core PUBLIC "${CODEX_GITX_SOURCE_DIR}/include"
    "${CODEX_GITX_SOURCE_DIR}/third_party/libgit2-1.8.4/include")
  target_link_libraries(codex-gitx-core PUBLIC libgit2package)
  target_link_libraries(codex-cpp PRIVATE codex-gitx-core)
  target_compile_definitions(codex-cpp PRIVATE CODEX_WITH_GITX=1)
  if(MSVC)
    target_compile_options(codex-gitx-core PRIVATE /utf-8)
  endif()
endif()
if(WIN32)
  target_link_libraries(codex-cpp PRIVATE ws2_32 bcrypt advapi32 shell32)
endif()
if(MSVC)
  target_compile_options(codex-cpp PRIVATE /utf-8 /W4)
else()
  target_compile_options(codex-cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_target(codex-web-assets ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/web $<TARGET_FILE_DIR:codex-cpp>/web
  DEPENDS codex-cpp
  VERBATIM)
