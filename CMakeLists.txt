cmake_minimum_required(VERSION 3.16)
project(CodexCpp LANGUAGES CXX)
option(CODEX_WITH_LIBGIT2 "Embedded libgit2 for web, collaboration, patches and Agent Git tools" ON)
find_package(Threads REQUIRED)
add_executable(codex-cpp codex.cpp)
target_compile_features(codex-cpp PRIVATE cxx_std_17)
set_target_properties(codex-cpp PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(codex-cpp PRIVATE Threads::Threads)
if(CODEX_WITH_LIBGIT2)
  enable_language(C)
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_CLI OFF CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(USE_SSH OFF CACHE STRING "" FORCE)
  set(USE_HTTPS OFF CACHE STRING "" FORCE)
  FetchContent_Declare(codex_libgit2
    URL https://codeload.github.com/libgit2/libgit2/tar.gz/refs/tags/v1.8.4
    URL_HASH SHA256=49d0fc50ab931816f6bfc1ac68f8d74b760450eebdb5374e803ee36550f26774)
  FetchContent_MakeAvailable(codex_libgit2)
  target_link_libraries(codex-cpp PRIVATE libgit2package)
  target_include_directories(codex-cpp PRIVATE "${codex_libgit2_SOURCE_DIR}/include")
  target_compile_definitions(codex-cpp PRIVATE CODEX_WITH_LIBGIT2=1)
endif()
if(WIN32)
  target_link_libraries(codex-cpp PRIVATE ws2_32 bcrypt advapi32 shell32)
endif()
if(MSVC)
  target_compile_options(codex-cpp PRIVATE /utf-8 /W4)
else()
  target_compile_options(codex-cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_target(codex-web-assets ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/web $<TARGET_FILE_DIR:codex-cpp>/web
  DEPENDS codex-cpp
  VERBATIM)
