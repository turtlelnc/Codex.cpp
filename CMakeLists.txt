cmake_minimum_required(VERSION 3.18)
project(CodexCpp LANGUAGES C CXX)
set(OPENSSL_USE_STATIC_LIBS TRUE)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
# Remove the obsolete cache switch: even old OFF caches now use the sole backend.
unset(CODEX_WITH_LIBGIT2 CACHE)
find_package(Threads REQUIRED)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(USE_SSH OFF CACHE STRING "" FORCE)
set(USE_HTTPS OFF CACHE STRING "" FORCE)
if(APPLE)
  # The OS supplies zlib. libgit2 1.8's bundled zlib misdetects modern Apple SDKs
  # as classic Mac OS and defines fdopen before the system stdio declaration.
  set(USE_BUNDLED_ZLIB OFF CACHE STRING "" FORCE)
else()
  set(USE_BUNDLED_ZLIB ON CACHE STRING "" FORCE)
endif()
FetchContent_Declare(codex_libgit2
  URL https://codeload.github.com/libgit2/libgit2/tar.gz/refs/tags/v1.8.4
  URL_HASH SHA256=49d0fc50ab931816f6bfc1ac68f8d74b760450eebdb5374e803ee36550f26774)
FetchContent_MakeAvailable(codex_libgit2)

# HTTP is part of the application; no curl executable or third-party DLLs.
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_CURL_MANUAL OFF CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
foreach(protocol DICT FILE FTP GOPHER IMAP LDAP LDAPS MQTT POP3 RTSP SMB SMTP TELNET TFTP)
  set(CURL_DISABLE_${protocol} ON CACHE BOOL "" FORCE)
endforeach()
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
set(CURL_STATIC_CRT ON CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS TRUE)
if(WIN32)
  set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
  set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
elseif(APPLE)
  set(USE_APPLE_SECTRUST ON CACHE BOOL "" FORCE)
  set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
else()
  set(OPENSSL_USE_STATIC_LIBS TRUE)
  set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
endif()
FetchContent_Declare(codex_curl
  URL https://codeload.github.com/curl/curl/tar.gz/refs/tags/curl-8_23_0
  URL_HASH SHA256=d0605d5883124ef5ae64c6d2b307e1417c482b362583b57491fdf37d9a99a792)
FetchContent_MakeAvailable(codex_curl)
include(cmake/Resources.cmake)
add_library(codex-resources STATIC "${CMAKE_CURRENT_BINARY_DIR}/codex_resources.cpp")
target_compile_features(codex-resources PUBLIC cxx_std_17)
target_include_directories(codex-resources PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

add_library(codex-runtime INTERFACE)
target_compile_features(codex-runtime INTERFACE cxx_std_17)
target_include_directories(codex-runtime INTERFACE "${codex_libgit2_SOURCE_DIR}/include")
target_link_libraries(codex-runtime INTERFACE Threads::Threads libgit2package CURL::libcurl codex-resources ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(codex-runtime INTERFACE -static-libstdc++ -static-libgcc)
endif()
if(WIN32)
  target_link_libraries(codex-runtime INTERFACE ws2_32 bcrypt advapi32 shell32)
endif()
add_executable(codex-cpp codex.cpp)
# Build this test target explicitly; no extra application-sized compile for users.
add_executable(codex-self-test EXCLUDE_FROM_ALL tests/agent_self_test.cpp)
add_executable(codex-transport-test EXCLUDE_FROM_ALL tests/http_transport_test.cpp)
foreach(target codex-cpp codex-self-test codex-transport-test)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  target_link_libraries(${target} PRIVATE codex-runtime)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_custom_target(codex-web-assets ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/web $<TARGET_FILE_DIR:codex-cpp>/web
  DEPENDS codex-cpp
  VERBATIM)

# Optional development modules/tests are never built or shipped by default.
add_executable(penguin-runtime-test EXCLUDE_FROM_ALL tests/penguin_runtime_test.cpp)
target_compile_features(penguin-runtime-test PRIVATE cxx_std_17)
target_link_libraries(penguin-runtime-test PRIVATE Threads::Threads)
add_library(penguin-demo MODULE EXCLUDE_FROM_ALL examples/penguin/native.cpp)
target_compile_features(penguin-demo PRIVATE cxx_std_17)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_sources(penguin-demo PRIVATE examples/penguin/native.def)
endif()
set_target_properties(penguin-demo PROPERTIES PREFIX "" OUTPUT_NAME "penguin-demo" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/penguin-demo" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/penguin-demo")
if(APPLE)
  set_target_properties(penguin-demo PROPERTIES SUFFIX ".dylib")
endif()
add_custom_command(TARGET penguin-demo POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/examples/penguin/penguin.json $<TARGET_FILE_DIR:penguin-demo>/penguin.json
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/examples/penguin/client.js $<TARGET_FILE_DIR:penguin-demo>/client.js)
