cmake_minimum_required(VERSION 3.18)
project(PenguinPlugin LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(penguin_plugin MODULE plugin.cpp)
target_compile_features(penguin_plugin PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(penguin_plugin PRIVATE /utf-8)
endif()
set_target_properties(penguin_plugin PROPERTIES PREFIX "" OUTPUT_NAME "@PENGUIN_ID@" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(config DEBUG RELEASE MINSIZEREL RELWITHDEBINFO)
  set_target_properties(penguin_plugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_sources(penguin_plugin PRIVATE native.def)
endif()
if(APPLE)
  set_target_properties(penguin_plugin PROPERTIES SUFFIX ".dylib")
  add_custom_command(TARGET penguin_plugin POST_BUILD COMMAND /usr/bin/codesign --force --sign - $<TARGET_FILE:penguin_plugin>)
endif()
